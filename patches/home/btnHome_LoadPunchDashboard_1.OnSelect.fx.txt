                                        StatusCode,
                                        StatusRows
                                    ),

                                    StatusName,
                                        First(StatusRows).StatusName,

                                    StatusOrder,
                                        First(StatusRows).StatusOrder,

                                    StatusColor,
                                        First(StatusRows).StatusColor,

                                    PunchCount,
                                        Sum(
                                            StatusRows,
                                            PunchCount
                                        )
                                ),

                                "StatusOrder",
                                SortOrder.Ascending
                            )
                        );

                        // =====================================================
                        // DEFAULT ANALYTICAL SELECTION
                        // =====================================================

                        ClearCollect(
                            colPunchExecutiveSelection,

                            {
                                SelectedRow:
                                    "All subcontractors",

                                SelectedColumn:
                                    "All categories",

                                PunchCount:
                                    Sum(
                                        colPunchExecutiveHeatmapCells,
                                        PunchCount
                                    ),

                                PercentOfTotal:
                                    100,

                                PercentOfRow:
                                    100,

                                IsSelection:
                                    false
                            }
                        )
                    );

                    true
                ),

                Set(
                    varPunchDashboardError,
                    FirstError.Message
                );

                false
            )
        );

        // =====================================================
        // FINAL SNAPSHOT STATE
        // =====================================================

        Set(
            varPunchDashboardSnapshotId,

            If(
                CountRows(
                    colPunchDashboardSnapshotInfo
                ) > 0,

                First(
                    colPunchDashboardSnapshotInfo
                ).SnapshotRunId,

                Blank()
            )
        );

        Set(
            varPunchDashboardLastRefresh,

            If(
                CountRows(
                    colPunchDashboardSnapshotInfo
                ) > 0,

                First(
                    colPunchDashboardSnapshotInfo
                ).GeneratedOn,

                Blank()
            )
        );

        Set(
            varPunchDashboardLoaded,
            IsBlank(varPunchDashboardError)
        );

        Set(
            varPunchDashboardLoading,
            false
        );

        Set(
            varExecutiveDashboardCompletedAt,
            Now()
        );

        Set(
            varExecutiveDashboardStatus,

            If(
                IsBlank(varPunchDashboardError),
                "READY",
                "ERROR"
            )
        );

        Set(
            varExecutiveDashboardError,
            Coalesce(
                varPunchDashboardError,
                ""
            )
        );

        If(
            !IsBlank(varPunchDashboardError),

            Notify(
                "Punch Dashboard could not be loaded: " &
                varPunchDashboardError,

                NotificationType.Error
            )
        );

        // =====================================================
        // RELEASE GLOBAL LOADING STATE
        // =====================================================

        Set(varGlobalLoading, false);
        Set(varGlobalLoadingMsg, "");
        Set(varGlobalLoadingSubMsg, "");
        Set(varGlobalLoadingScope, "");
        Set(varGlobalLoadAction, "");
        Set(varGlobalLoadPending, false);

        Set(varPunchDashboardLoading, false);
        Set(varProjectSwitching, false);
    )
)
