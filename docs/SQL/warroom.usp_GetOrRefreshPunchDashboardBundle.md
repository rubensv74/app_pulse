/*
===============================================================================
Procedure:
    warroom.usp_GetOrRefreshPunchDashboardBundle

Purpose:
    Returns the Punch Dashboard bundle for a project and template.

    If no completed snapshot exists, or the latest completed snapshot is older
    than the configured maximum age, the procedure first generates a new
    snapshot and then returns the dashboard bundle.

Dependencies:
    warroom.usp_GeneratePunchDashboardSnapshot
    warroom.usp_GetPunchDashboardBundle
    warroom.PunchDashboardSnapshotRun

Return:
    One result set with one column:

        result NVARCHAR(MAX)

    The result column contains the JSON contract generated by
    warroom.usp_GetPunchDashboardBundle.

Notes:
    - @ForceRefresh = 1 forces a new snapshot.
    - @MaxSnapshotAgeMinutes = 0 also forces a new snapshot.
    - The output produced by usp_GeneratePunchDashboardSnapshot is captured
      internally so Power Automate receives only the final dashboard result.
===============================================================================
*/

CREATE OR ALTER PROCEDURE warroom.usp_GetOrRefreshPunchDashboardBundle
(
    @ProjectId             BIGINT,
    @TemplateId            BIGINT,
    @RequestedBy           NVARCHAR(450) = NULL,
    @ForceRefresh          BIT = 0,
    @MaxSnapshotAgeMinutes INT = 30,
    @KeepCompletedRuns     INT = 7
)
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;

    ---------------------------------------------------------------------------
    -- Input validation
    ---------------------------------------------------------------------------
    IF @ProjectId IS NULL OR @ProjectId <= 0
        THROW 50200, 'ProjectId must be greater than zero.', 1;

    IF @TemplateId IS NULL OR @TemplateId <= 0
        THROW 50201, 'TemplateId must be greater than zero.', 1;

    IF @MaxSnapshotAgeMinutes IS NULL
        SET @MaxSnapshotAgeMinutes = 30;

    IF @MaxSnapshotAgeMinutes < 0
        SET @MaxSnapshotAgeMinutes = 0;

    IF @KeepCompletedRuns IS NULL OR @KeepCompletedRuns < 1
        SET @KeepCompletedRuns = 1;

    IF @KeepCompletedRuns > 20
        SET @KeepCompletedRuns = 20;

    ---------------------------------------------------------------------------
    -- Read latest completed snapshot
    ---------------------------------------------------------------------------
    DECLARE
        @LatestSnapshotRunId BIGINT,
        @LatestCompletedOn   DATETIME2(7),
        @SnapshotAgeMinutes  INT,
        @MustGenerate        BIT = 0;

    SELECT TOP (1)
        @LatestSnapshotRunId = r.SnapshotRunId,
        @LatestCompletedOn = r.CompletedOn
    FROM warroom.PunchDashboardSnapshotRun r
    WHERE r.ProjectId = @ProjectId
      AND r.TemplateId = @TemplateId
      AND r.Status = 'COMPLETED'
    ORDER BY r.SnapshotRunId DESC;

    IF @LatestCompletedOn IS NOT NULL
    BEGIN
        SET @SnapshotAgeMinutes =
            DATEDIFF
            (
                MINUTE,
                @LatestCompletedOn,
                SYSUTCDATETIME()
            );
    END;

    ---------------------------------------------------------------------------
    -- Decide whether the snapshot must be generated
    ---------------------------------------------------------------------------
    IF COALESCE(@ForceRefresh, 0) = 1
        SET @MustGenerate = 1;

    IF @LatestSnapshotRunId IS NULL
        SET @MustGenerate = 1;

    IF @LatestSnapshotRunId IS NOT NULL
       AND COALESCE(@SnapshotAgeMinutes, 2147483647)
           >= @MaxSnapshotAgeMinutes
    BEGIN
        SET @MustGenerate = 1;
    END;

    ---------------------------------------------------------------------------
    -- Generate snapshot when required
    --
    -- Capture the result set from usp_GeneratePunchDashboardSnapshot so that
    -- the caller receives only the final dashboard bundle result set.
    ---------------------------------------------------------------------------
    IF @MustGenerate = 1
    BEGIN
        DECLARE @GenerationResult TABLE
        (
            Success           BIT,
            SnapshotRunId     BIGINT,
            ProjectId         BIGINT,
            TemplateId        BIGINT,
            SnapshotStatus    VARCHAR(20),
            SourcePunchCount  BIGINT,
            GeneratedOn       DATETIME2(7),
            DurationMs        BIGINT,
            [Message]         NVARCHAR(4000)
        );

        INSERT INTO @GenerationResult
        (
            Success,
            SnapshotRunId,
            ProjectId,
            TemplateId,
            SnapshotStatus,
            SourcePunchCount,
            GeneratedOn,
            DurationMs,
            [Message]
        )
        EXEC warroom.usp_GeneratePunchDashboardSnapshot
            @ProjectId = @ProjectId,
            @TemplateId = @TemplateId,
            @RequestedBy = @RequestedBy,
            @KeepCompletedRuns = @KeepCompletedRuns;

        IF NOT EXISTS
        (
            SELECT 1
            FROM @GenerationResult gr
            WHERE gr.Success = 1
              AND gr.SnapshotStatus = 'COMPLETED'
        )
        BEGIN
            THROW 50202, 'The Punch Dashboard snapshot was not completed.', 1;
        END;
    END;

    ---------------------------------------------------------------------------
    -- Return the dashboard JSON bundle
    ---------------------------------------------------------------------------
    EXEC warroom.usp_GetPunchDashboardBundle
        @ProjectId = @ProjectId,
        @TemplateId = @TemplateId;
END;
GO
